Relational View
WITH RegionalTotals AS (
  SELECT rep_id, manager_name, region, SUM(deal_amount) AS total_revenue,
         DENSE_RANK() OVER(PARTITION BY region ORDER BY SUM(deal_amount) DESC) AS regional_rank
  FROM sales_transactions INNER JOIN sales_reps ON sales_transactions.rep_id = sales_reps.id
  GROUP BY rep_id, manager_name, region
)
SELECT * FROM RegionalTotals WHERE regional_rank <= 3;
